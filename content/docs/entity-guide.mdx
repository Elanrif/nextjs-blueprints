## 🎯 Pattern {entity}

Le template fournit un **pattern générique réutilisable** pour toute entité métier. En remplaçant `{entity}` par votre nom (ex: `users`, `products`, `orders`), vous obtenez automatiquement :

✅ Types TypeScript  
✅ Schémas de validation Zod  
✅ Services API centralisés  
✅ Hooks React Query  
✅ Mutations avec cache invalidation  
✅ Formulaires réactifs  
✅ Tables de données

---

## 📁 Structure pour une Entité

```
lib/{entity}/
├── api/
│   ├── types.ts              # Types TypeScript
│   ├── mutations.ts          # useMutation hooks
│   ├── action.ts             # Server actions
│   ├── queries/
│   │   ├── index.ts
│   │   ├── queries.client.ts # useQuery hooks
│   │   └── queries.server.ts # Requêtes serveur
│   └── services/
│       ├── {entity}.client.ts
│       └── {entity}.server.ts
├── components/
│   ├── {entity}-form.tsx
│   ├── {entity}-listing.tsx
│   ├── {entity}-view-page.tsx
│   └── {entity}-table/
│       └── columns.tsx
└── schemas/
    └── {entity}.schema.ts
```

## 📝 Exemple Complet : Créer une Entité `users`

### 1. Types TypeScript

**File:** `lib/users/api/types.ts`

```typescript
/**
 * @file types.ts
 * @description Types pour l'entité User
 */

export interface User {
  id: number;
  name: string;
  email: string;
  role: "admin" | "user" | "guest";
  createdAt: string;
  updatedAt: string;
}

export type UserCreate = Omit<User, "id" | "createdAt" | "updatedAt">;
export type UserUpdate = Partial<UserCreate>;

export interface UserFilters {
  page?: number;
  size?: number;
  search?: string;
  role?: string;
}

export interface UserResponse {
  data: User[];
  total: number;
  page: number;
  size: number;
  totalPages: number;
}
```

### 2. Schéma de Validation

**File:** `lib/users/schemas/user.schema.ts`

```typescript
import { z } from "zod";

export const UserCreateSchema = z.object({
  name: z.string().min(1, "Le nom est requis").max(100, "Max 100 caractères"),
  email: z.string().email("Email invalide"),
  role: z.enum(["admin", "user", "guest"]),
});

export const UserUpdateSchema = UserCreateSchema.partial();

export type UserCreate = z.infer<typeof UserCreateSchema>;
export type UserUpdate = z.infer<typeof UserUpdateSchema>;
```

### 3. Service API

**File:** `lib/users/api/services/user.client.ts`

```typescript
import { axiosInstance } from "@/lib/config/axios.config";
import {
  User,
  UserCreate,
  UserUpdate,
  UserResponse,
  UserFilters,
} from "../types";

class UserClientService {
  private readonly endpoint = "/users";

  async list(filters?: UserFilters): Promise<UserResponse> {
    const { data } = await axiosInstance.get<UserResponse>(this.endpoint, {
      params: {
        page: filters?.page ?? 1,
        size: filters?.size ?? 10,
        ...filters,
      },
    });
    return data;
  }

  async get(id: number): Promise<User> {
    const { data } = await axiosInstance.get<User>(`${this.endpoint}/${id}`);
    return data;
  }

  async create(payload: UserCreate): Promise<User> {
    const { data } = await axiosInstance.post<User>(this.endpoint, payload);
    return data;
  }

  async update(id: number, payload: UserUpdate): Promise<User> {
    const { data } = await axiosInstance.patch<User>(
      `${this.endpoint}/${id}`,
      payload,
    );
    return data;
  }

  async delete(id: number): Promise<void> {
    await axiosInstance.delete(`${this.endpoint}/${id}`);
  }
}

export const userService = new UserClientService();
```

### 4. React Query Hooks

**File:** `lib/users/api/queries/queries.client.ts`

```typescript
import { useQuery } from "@tanstack/react-query";
import { userService } from "../services/user.client";
import { User, UserResponse, UserFilters } from "../types";

export const useUserList = (filters?: UserFilters) => {
  return useQuery<UserResponse>({
    queryKey: ["users:list", filters],
    queryFn: () => userService.list(filters),
    staleTime: 5 * 60 * 1000, // 5 minutes
  });
};

export const useUser = (id: number | null) => {
  return useQuery<User>({
    queryKey: ["users:detail", id],
    queryFn: () => userService.get(id!),
    enabled: !!id,
  });
};
```

### 5. Mutations

**File:** `lib/users/api/mutations.ts`

```typescript
import { useMutation, useQueryClient } from "@tanstack/react-query";
import { userService } from "./services/user.client";
import { User, UserCreate, UserUpdate } from "./types";
import { toast } from "sonner";

export const useCreateUser = () => {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: userService.create,
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["users:list"] });
      toast.success("Utilisateur créé");
    },
    onError: (error: any) => {
      toast.error(error?.message || "Erreur");
    },
  });
};

export const useUpdateUser = (id: number) => {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: (data: UserUpdate) => userService.update(id, data),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["users:list"] });
      queryClient.invalidateQueries({ queryKey: ["users:detail", id] });
      toast.success("Utilisateur mis à jour");
    },
  });
};

export const useDeleteUser = (id: number) => {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: () => userService.delete(id),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["users:list"] });
      toast.success("Utilisateur supprimé");
    },
  });
};
```

### 6. Formulaire

**File:** `lib/users/components/user-form.tsx`

```typescript
"use client";

import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import { UserCreateSchema, UserCreate } from "../schemas/user.schema";
import { useCreateUser } from "../api/mutations";

export function UserForm() {
  const form = useForm<UserCreate>({
    resolver: zodResolver(UserCreateSchema),
    defaultValues: {
      name: "",
      email: "",
      role: "user",
    },
  });

  const mutation = useCreateUser();

  const onSubmit = async (data: UserCreate) => {
    await mutation.mutateAsync(data);
    form.reset();
  };

  return (
    <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
      <div>
        <label>Nom</label>
        <input
          {...form.register("name")}
          type="text"
          className="border rounded px-2 py-1"
        />
        {form.formState.errors.name && (
          <p className="text-red-500">{form.formState.errors.name.message}</p>
        )}
      </div>

      <div>
        <label>Email</label>
        <input
          {...form.register("email")}
          type="email"
          className="border rounded px-2 py-1"
        />
        {form.formState.errors.email && (
          <p className="text-red-500">{form.formState.errors.email.message}</p>
        )}
      </div>

      <div>
        <label>Rôle</label>
        <select {...form.register("role")} className="border rounded px-2 py-1">
          <option value="user">Utilisateur</option>
          <option value="admin">Admin</option>
          <option value="guest">Invité</option>
        </select>
      </div>

      <button
        type="submit"
        disabled={mutation.isPending}
        className="bg-blue-500 text-white px-4 py-2 rounded"
      >
        {mutation.isPending ? "Création..." : "Créer"}
      </button>
    </form>
  );
}
```

### 7. Table de Données

**File:** `lib/users/components/user-table/columns.tsx`

import { ColumnDef } from "@tanstack/react-table";
import { User } from "../../api/types";
import { useDeleteUser } from "../../api/mutations";

export const userColumns: ColumnDef<User>[] = [
  {
    accessorKey: "name",
    header: "Nom",
  },
  {
    accessorKey: "email",
    header: "Email",
  },
  {
    accessorKey: "role",
    header: "Rôle",
    cell: ({ row }) => (
      <span className="capitalize">{row.original.role}</span>
    ),
  },
  {
    accessorKey: "createdAt",
    header: "Créé",
    cell: ({ row }) => new Date(row.original.createdAt).toLocaleDateString("fr"),
  },
  {
    id: "actions",
    cell: ({ row }) => {
      const mutation = useDeleteUser(row.original.id);
      return (
        <button
          onClick={() => mutation.mutate()}
          className="text-red-500 hover:underline"
        >
          Supprimer
        </button>
      );
    },
  },
];

## 🚀 Utilisation dans une Page

**File:** `src/app/users/page.tsx`

```typescript
"use client";

import { useUserList } from "@/lib/users/api/queries";
import { UserForm } from "@/lib/users/components/user-form";
import { userColumns } from "@/lib/users/components/user-table/columns";
import { DataTable } from "@/lib/_/components/ui/data-table";

export default function UsersPage() {
  const { data, isLoading, error } = useUserList({ page: 1, size: 20 });

  return (
    <div className="space-y-6 p-6">
      <h1 className="text-3xl font-bold">Utilisateurs</h1>

      <UserForm />

      {isLoading && <p>Chargement...</p>}
      {error && <p className="text-red-500">Erreur : {error.message}</p>}
      {data && (
        <DataTable columns={userColumns} data={data.data} />
      )}
    </div>
  );
}
```

## ✅ Checklist Entité

Pour chaque nouvelle entité :

- [ ] Dossier `lib/{entity}/` créé
- [ ] Types définis dans `types.ts`
- [ ] Schéma Zod dans `schemas/{entity}.schema.ts`
- [ ] Service dans `services/{entity}.client.ts`
- [ ] Queries hooks dans `queries/queries.client.ts`
- [ ] Mutations dans `mutations.ts`
- [ ] Formulaire dans `components/{entity}-form.tsx`
- [ ] Colonnes table dans `components/{entity}-table/columns.tsx`
- [ ] Page créée dans `src/app/{entity}/page.tsx`

## 💡 Bonnes Pratiques

### 1. Toujours valider

```typescript
// ✅ BON
const validData = UserCreateSchema.parse(data);

// ❌ MAUVAIS
const data: User = data as User;
```

### 2. Utiliser les services

```typescript
// ✅ BON
const { data } = useUserList();

// ❌ MAUVAIS
const [data, setData] = useState([]);
useEffect(() => {
  fetch("/api/users")
    .then((r) => r.json())
    .then(setData);
}, []);
```

### 3. Cache invalidation

```typescript
// ✅ BON - Après une mutation
onSuccess: () => {
  queryClient.invalidateQueries({ queryKey: ["users:list"] });
},

// ❌ MAUVAIS - Pas d'invalidation
onSuccess: () => {
  // ...
},
```
